cmake_minimum_required(VERSION 3.20)
project(driver2_browser LANGUAGES C CXX)
file(GLOB_RECURSE GAME CONFIGURE_DEPENDS Game/*.c utils/*.cpp utils/*.c PsyCross/src/*.c PsyCross/src/*.C PsyCross/src/*.cpp)
add_executable(driver2 ${GAME} redriver2_psxpc.cpp platform/Emscripten/emscripten_exports.cpp)
set_source_files_properties(${GAME} PROPERTIES LANGUAGE CXX)
target_include_directories(driver2 PRIVATE Game . PsyCross/include PsyCross/include/psx)
target_compile_definitions(driver2 PRIVATE NTSC_VERSION NDEBUG BUILD_CONFIGURATION_STRING="Release")
target_compile_options(driver2 PRIVATE -O2 -sUSE_SDL=2 -sUSE_LIBJPEG=1 -Wno-c++11-narrowing -Wno-constant-conversion -Wno-writable-strings -Wno-format -Wno-deprecated)
target_link_options(driver2 PRIVATE -sUSE_SDL=2 -sUSE_LIBJPEG=1 -sFULL_ES2=1 -sASYNCIFY -sASYNCIFY_STACK_SIZE=131072 -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=268435456 -sFORCE_FILESYSTEM=1 -sEXPORTED_RUNTIME_METHODS=FS,IDBFS,callMain,ccall,writeArrayToMemory -sEXPORTED_FUNCTIONS=_main,_malloc -lidbfs.js -lopenal -sINCOMING_MODULE_JS_API=arguments,canvas,locateFile,preRun,onRuntimeInitialized,onAbort,print,printErr,noInitialRun)
set_target_properties(driver2 PROPERTIES SUFFIX .js)

if(PEAK_DRIVER_DATA)
 target_link_options(driver2 PRIVATE "--preload-file" "${PEAK_DRIVER_DATA}@/game")
endif()
