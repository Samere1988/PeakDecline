cmake_minimum_required(VERSION 3.24)
project(FZeroBrowser LANGUAGES C CXX)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "This target requires the Emscripten toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_compile_options(-pthread -sUSE_SDL=2 -sUSE_ZLIB=1 -ffp-contract=off -fexceptions)
add_compile_definitions(FMT_CONSTEVAL=)
add_link_options(-pthread -sUSE_SDL=2 -sUSE_ZLIB=1 -fexceptions)
include(FetchContent)
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json)
FetchContent_Declare(spdlog URL https://github.com/gabime/spdlog/archive/refs/tags/v1.14.1.tar.gz)
FetchContent_MakeAvailable(spdlog)
FetchContent_Declare(tinyxml2 URL https://github.com/leethomason/tinyxml2/archive/refs/tags/10.0.0.tar.gz)
set(tinyxml2_BUILD_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(tinyxml2)

# Use Emscripten's zlib and SDL ports, not Windows host libraries.
add_library(ZLIB::ZLIB INTERFACE IMPORTED GLOBAL)
set_target_properties(ZLIB::ZLIB PROPERTIES INTERFACE_LINK_OPTIONS "-sUSE_ZLIB=1")
set(ZLIB_FOUND TRUE)
set(ZLIB_VERSION_STRING 1.3.1)
add_library(SDL2::SDL2 INTERFACE IMPORTED GLOBAL)
set_target_properties(SDL2::SDL2 PROPERTIES INTERFACE_COMPILE_OPTIONS "-sUSE_SDL=2" INTERFACE_LINK_OPTIONS "-sUSE_SDL=2")
FetchContent_Declare(libzip URL https://github.com/nih-at/libzip/archive/refs/tags/v1.11.3.tar.gz)
foreach(option ENABLE_BZIP2 ENABLE_LZMA ENABLE_ZSTD ENABLE_OPENSSL ENABLE_GNUTLS ENABLE_MBEDTLS ENABLE_WINDOWS_CRYPTO BUILD_TOOLS BUILD_REGRESS BUILD_EXAMPLES BUILD_DOC)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
FetchContent_MakeAvailable(libzip)
if(NOT TARGET libzip::zip)
  add_library(libzip::zip ALIAS zip)
endif()

set(USE_OPENGLES ON CACHE BOOL "" FORCE)
set(DISABLE_DLL_LOADER ON CACHE BOOL "" FORCE)
set(ENABLE_SCRIPTING OFF CACHE BOOL "" FORCE)
set(INCLUDE_MPQ_SUPPORT OFF CACHE BOOL "" FORCE)
set(PRISM_STANDALONE OFF CACHE BOOL "" FORCE)
add_compile_definitions(ENABLE_OPENGL USE_OPENGLES IMGUI_IMPL_OPENGL_ES3)
add_subdirectory("${BROWSER_LUS}" lus)
target_link_libraries(libultraship PUBLIC SDL2::SDL2)
target_link_libraries(ImGui PUBLIC SDL2::SDL2)

# Link the renderer first so platform errors are found independently of game glue.
add_executable(renderer_probe renderer_probe.cpp ../src/cartridge_features.cpp)
target_link_libraries(renderer_probe PRIVATE libultraship)
target_link_options(renderer_probe PRIVATE -sUSE_WEBGL2=1 -sFULL_ES3=1
  -sALLOW_MEMORY_GROWTH=1 -sPTHREAD_POOL_SIZE=4 -sASSERTIONS=1 -g3)
target_link_options(renderer_probe PRIVATE "--preload-file=${CMAKE_BINARY_DIR}/gdiffuser.o2r@/gdiffuser.o2r")
include(game.cmake)
